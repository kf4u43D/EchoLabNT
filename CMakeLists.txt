file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" ECHOLAB_PROJECT_VERSION LIMIT_COUNT 1)
if(NOT ECHOLAB_PROJECT_VERSION MATCHES "^([0-9]+\\.[0-9]+\\.[0-9]+)(-[0-9A-Za-z.-]+)?$")
    message(FATAL_ERROR "Invalid VERSION")

set(ECHOLAB_NUMERIC_VERSION "${CMAKE_MATCH_1}")

set(CMAKE_CXX_STANDARD 17)

set(CMAKE_CXX_EXTENSIONS OFF)

set(ECHOLAB_BUILD_ID "" CACHE STRING "Override build ID; empty derives version plus source SHA-256")
set(echolab_source_signature "")
foreach(source IN ITEMS VERSION CMakeLists.txt src/dsp/engine.h src/dsp/engine.cpp src/dsp/segment_models.h src/dsp/rack_models.h src/dsp/output_limiter.h src/dsp/loop_exciter.h src/dsp/multitap_patterns.h src/dsp/analog_color.h src/dsp/a04_color.h src/dsp/legacy_voicings.h src/host/wav.h src/host/wav.cpp src/host/sha256.h src/host/sha256.cpp src/host/renderer.cpp src/host/profile_timing.h src/nt/clock_sync.h)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" source_sha256)
    string(APPEND echolab_source_signature "${source}:${source_sha256}\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
endforeach()
string(SHA256 ECHOLAB_SOURCE_SHA256 "${echolab_source_signature}")
string(SUBSTRING "${ECHOLAB_SOURCE_SHA256}" 0 12 echolab_source_short)
if(ECHOLAB_BUILD_ID STREQUAL "" OR ECHOLAB_BUILD_ID STREQUAL "source-unversioned")
    set(ECHOLAB_EFFECTIVE_BUILD_ID "${ECHOLAB_PROJECT_VERSION}+${echolab_source_short}")
else()
    set(ECHOLAB_EFFECTIVE_BUILD_ID "${ECHOLAB_BUILD_ID}")
endif()
message(STATUS "EchoLabNT host build ID: ${ECHOLAB_EFFECTIVE_BUILD_ID}")

function(echolab_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(ECHOLAB_SANITIZERS)
        if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" AND NOT MSVC)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        else()
            message(FATAL_ERROR "ECHOLAB_SANITIZERS needs GCC/Clang with address/undefined sanitizer runtime")
        endif()
    endif()
    if(ECHOLAB_MSVC_ASAN)
        if(MSVC)
            target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
            target_link_options(${target} PRIVATE /INCREMENTAL:NO)
        else()
            message(FATAL_ERROR "ECHOLAB_MSVC_ASAN requires the MSVC toolchain; GCC/Clang use ECHOLAB_SANITIZERS")
        endif()
    endif()
endfunction()

add_library(echolab_dsp STATIC src/dsp/engine.cpp)
target_include_directories(echolab_dsp PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
echolab_warnings(echolab_dsp)

add_library(echolab_wav STATIC src/host/wav.cpp src/host/sha256.cpp)
target_include_directories(echolab_wav PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src/host")
echolab_warnings(echolab_wav)

add_executable(echo_render src/host/renderer.cpp)
set_target_properties(echo_render PROPERTIES OUTPUT_NAME echolabnt_render)
target_link_libraries(echo_render PRIVATE echolab_dsp echolab_wav)
target_compile_definitions(echo_render PRIVATE ECHOLAB_BUILD_ID="${ECHOLAB_EFFECTIVE_BUILD_ID}"
    ECHOLAB_SOURCE_SHA256="${ECHOLAB_SOURCE_SHA256}"
    ECHOLAB_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}"
    ECHOLAB_CONFIGURATION="$<CONFIG>")
if(ECHOLAB_MSVC_ASAN)
    target_compile_definitions(echo_render PRIVATE ECHOLAB_SANITIZER="address")
elseif(ECHOLAB_SANITIZERS)
    target_compile_definitions(echo_render PRIVATE ECHOLAB_SANITIZER="address,undefined")
else()
    target_compile_definitions(echo_render PRIVATE ECHOLAB_SANITIZER="none")
endif()
echolab_warnings(echo_render)
